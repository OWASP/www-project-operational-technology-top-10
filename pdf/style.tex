%   /work/owasp-white.pdf   OWASP mark, white

% ---------------------------------------------------------------------------
% Full-page artwork without a second LaTeX run: the shipout/background hook
% draws #1 behind the next page, in a \paperwidth x \paperheight TikZ canvas
% whose origin is the bottom-left corner.
\newcommand\ot@background[1]{%
  \AddToHookNext{shipout/background}{%
    \put(0,0){\raisebox{-\paperheight}[0pt][0pt]{%
      \begin{tikzpicture}[x=1pt,y=1pt]
        \useasboundingbox (0,0) rectangle (\paperwidth,\paperheight);
        #1
      \end{tikzpicture}}}}}

% The OWASP lockup: mark plus wordmark, white, for the blue pages.
\newcommand\ot@owasplockup[1]{% #1: mark height
  \raisebox{-.22\height}{\includegraphics[height=#1]{/work/owasp-white.pdf}}%
  \hspace{.3em}{\otlight\fontsize{#1}{#1}\selectfont OWASP}}

% ---------------------------------------------------------------------------
% Cover. #1 edition year, #2 project leaders separated by \\.
\newcommand\otcover[2]{%
  \begin{titlepage}
  \ot@background{%
    \fill[otblue] (0,0) rectangle (\paperwidth,\paperheight);
    % Oversized emblem bleeding off the right edge, as a watermark.
    \node[inner sep=0] at (.80\paperwidth,.38\paperheight)
      {\includegraphics[width=.80\paperwidth]{/work/emblem-watermark.pdf}};
    \fill[otnavy] (0,0) rectangle (\paperwidth,.19\paperheight);
    \node[anchor=north west, inner sep=0, text=white]
      at (22mm,\paperheight-22mm) {\ot@owasplockup{9mm}};
    % Multi-line nodes get a text width: without one TikZ sets align= as an
    % \halign, which breaks on font groups that span a \\.
    \node[anchor=north west, inner sep=0, text=white, align=left,
          text width=.75\paperwidth]
      at (22mm,.72\paperheight) {%
        % Black weight sets its word spaces too tight at display size.
        {\otblack\fontsize{86}{80}\selectfont OT\hspace{.25em}Top\hspace{.25em}10\par}\vspace{7mm}
        {\otlight\fontsize{27}{33}\selectfont The Top 10 Operational\\
         Technology Security Risks\par}\vspace{9mm}
        {\otsemibold\fontsize{15}{18}\selectfont\color{otpale}#1 Edition\par}};
    \node[anchor=south west, inner sep=0, text=white, align=left,
          text width=.5\paperwidth]
      at (22mm,22mm) {%
        {\otsemibold\fontsize{9}{11}\selectfont\color{otpale}PROJECT LEADERS\par}\vspace{2.5mm}
        {\fontsize{11.5}{15}\selectfont #2\par}};
    \node[anchor=south east, inner sep=0, text=white]
      at (\paperwidth-22mm,22mm) {\fontsize{11.5}{15}\selectfont ot.owasp.org};
  }%
  \null
  \end{titlepage}}
